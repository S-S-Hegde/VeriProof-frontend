// Function to set styles for a given progress value
function setNavStyles(progress) {
  if (linksBlock) {
    const maxGap = 2.5 // 2.5vw
    const minGap = 1 // 1vw
    const gapVW = maxGap - (maxGap - minGap) * progress
    linksBlock.style.gap = `${gapVW}vw`
  }

  desktopNavLinks.forEach((link) => {
    const maxPadding = 1.5 // 1.5vw
    const minPadding = 0.5 // 0.5vw
    const padVW = maxPadding - (maxPadding - minPadding) * progress
    link.style.padding = `inherit ${padVW}vw`
  })

  if (navBDesktop) {
    const maxGapB = 13 // 13vw
    const minGapB = 6 // 6vw
    const gapBWV = maxGapB - (maxGapB - minGapB) * progress
    navBDesktop.style.gap = `${gapBWV}vw`
  }

  if (navBrand) {
    const maxScale = 1.2
    const minScale = 1
    const scale = maxScale - (maxScale - minScale) * progress
    navBrand.style.transform = `scale(${scale})`
  }
}

      entries.forEach(entry => {

        }
      });
    }, {
      threshold: 0.6 // 50% visibility required
    });

    videos.forEach(video => observer.observe(video));
  });

const getDynamicSpacingSwiper1 = () => window.innerWidth * 0.01111;

document.addEventListener("DOMContentLoaded", function () {
  const videos = document.querySelectorAll('video');

  // Play all videos except .video-footer and .flowers-mob-video's video
  videos.forEach(video => {
    if (
      !video.classList.contains('video-footer') &&
      !video.closest('.flowers-mob-video')
    ) {
      video.play().catch(function(error) {
        console.warn("Autoplay blocked on load:", error);
      });
    }
  });

  // Rewind and replay on tab focus — excluding .video-footer and .flowers-mob-video
  document.addEventListener('visibilitychange', function () {
    if (!document.hidden) {
      videos.forEach(video => {
        if (
          !video.classList.contains('video-footer') &&
          !video.closest('.flowers-mob-video')
        ) {
          try {
            video.currentTime = 0;
            video.play();
          } catch (error) {
            console.warn("Error restarting video:", error);
          }
        }
      });
    }
  });
});

  // Link timelines to scroll position
  function createScrollTrigger(triggerElement, timeline) {
    // Reset tl when scroll out of view past bottom of screen
    ScrollTrigger.create({
      trigger: triggerElement,
      start: "top bottom",
      onLeaveBack: () => {
        timeline.progress(0);
        timeline.pause();
      }
    });
    // Play tl when scrolled into view (60% from top of screen)
    ScrollTrigger.create({
      trigger: triggerElement,
      start: "top 60%",
      onEnter: () => timeline.play()
    });
  }
